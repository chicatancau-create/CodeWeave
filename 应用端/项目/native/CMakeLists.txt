cmake_minimum_required(VERSION 3.25)
project(CodeWeaveNative LANGUAGES CXX)

option(QEWEAVE_BUILD_UI_QT "Build the optional Qt Quick desktop shell" OFF)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Ship the native candidate without a machine-wide VC++ runtime prerequisite.
# The Windows system UCRT/API sets remain OS components; MSVC's C++ runtime is
# linked into this executable so a portable copy can be launched directly.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_library(codeweave-core STATIC src/app_core.cpp src/account.cpp src/account_store.cpp src/account_service.cpp src/password_crypto.cpp src/content_contract.cpp src/content_repository.cpp src/content_access.cpp src/learning_contract.cpp src/mastery_contract.cpp src/mastery_store.cpp src/mastery_service.cpp src/review_scheduler.cpp src/operation_index.cpp src/evidence_store.cpp src/progress_store.cpp src/challenge_service.cpp src/external_evidence_service.cpp src/progress_projection.cpp src/review_service.cpp src/manifest_dry_run.cpp src/migration_dry_run.cpp src/migration_staging.cpp src/migration_contract.cpp src/legacy_import.cpp src/backup_service.cpp src/backend_task_service.cpp src/execution_contract.cpp src/process_contract.cpp src/execution_plan.cpp src/process_runner.cpp src/execution_service.cpp src/trace_contract.cpp src/trace_store.cpp)
target_include_directories(codeweave-core PUBLIC src)
target_compile_definitions(codeweave-core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
if (MSVC)
  # All project sources are UTF-8. Keep MSVC's source decoding independent of
  # the machine ACP (for example, code page 936 on Chinese Windows).
  target_compile_options(codeweave-core PRIVATE /utf-8)
endif()

add_executable(codeweave-native WIN32 src/main.cpp)
target_link_libraries(codeweave-native PRIVATE codeweave-core)
if (WIN32)
  target_link_libraries(codeweave-native PRIVATE comdlg32)
endif()
target_compile_definitions(codeweave-native PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if (MSVC)
  target_compile_options(codeweave-native PRIVATE /W4 /permissive- /EHsc /utf-8)
else()
  target_compile_options(codeweave-native PRIVATE -Wall -Wextra -Wpedantic)
  target_link_options(codeweave-native PRIVATE -static-libgcc -static-libstdc++ -static)
endif()
if (WIN32)
  target_link_libraries(codeweave-core PUBLIC bcrypt)
endif()

install(TARGETS codeweave-native RUNTIME DESTINATION .)

include(CTest)
if (BUILD_TESTING)
  add_executable(codeweave-native-tests src/native_tests.cpp)
  target_link_libraries(codeweave-native-tests PRIVATE codeweave-core)
  if (MSVC)
    target_compile_options(codeweave-native-tests PRIVATE /UNDEBUG /utf-8)
  else()
    target_compile_options(codeweave-native-tests PRIVATE -UNDEBUG)
  endif()
  add_test(NAME native-core COMMAND codeweave-native-tests)
  add_executable(codeweave-content-delivery-tests src/content_delivery_tests.cpp)
  target_link_libraries(codeweave-content-delivery-tests PRIVATE codeweave-core)
  if (MSVC)
    target_compile_options(codeweave-content-delivery-tests PRIVATE /UNDEBUG /utf-8)
  else()
    target_compile_options(codeweave-content-delivery-tests PRIVATE -UNDEBUG)
  endif()
  add_test(NAME native-content-delivery COMMAND codeweave-content-delivery-tests)
  add_test(NAME native-startup COMMAND codeweave-native --self-test)
  add_test(NAME native-gui-smoke COMMAND codeweave-native --gui-smoke)
  add_test(NAME native-gui-interaction-smoke COMMAND codeweave-native --gui-interaction-smoke)
  add_executable(codeweave-test-process-tool src/test_process_tool.cpp)
  if (MSVC)
    target_compile_options(codeweave-test-process-tool PRIVATE /utf-8)
  endif()
  set_target_properties(codeweave-test-process-tool PROPERTIES OUTPUT_NAME "codeweave-cpp-runner")
  add_dependencies(codeweave-native-tests codeweave-test-process-tool)
  add_executable(codeweave-backend-benchmark src/backend_benchmark.cpp)
  target_link_libraries(codeweave-backend-benchmark PRIVATE codeweave-core)
  if (MSVC)
    target_compile_options(codeweave-backend-benchmark PRIVATE /W4 /permissive- /EHsc /utf-8)
  else()
    target_compile_options(codeweave-backend-benchmark PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  find_program(CODEWEAVE_TEST_CPP_COMPILER NAMES g++ clang++ cl)
  find_program(CODEWEAVE_TEST_JAVA_RUNTIME NAMES java.exe java)
  find_program(CODEWEAVE_TEST_PYTHON_RUNTIME NAMES python.exe python)
  find_program(CODEWEAVE_TEST_NODE_RUNTIME NAMES node.exe node)
  if(CODEWEAVE_TEST_CPP_COMPILER)
    add_test(NAME native-cpp-compile-run
      COMMAND codeweave-native-tests $<TARGET_FILE:codeweave-test-process-tool>
      ${CODEWEAVE_TEST_CPP_COMPILER} ${CODEWEAVE_TEST_JAVA_RUNTIME}
      ${CODEWEAVE_TEST_PYTHON_RUNTIME} ${CODEWEAVE_TEST_NODE_RUNTIME})
  endif()
endif()

if (QEWEAVE_BUILD_UI_QT)
  add_subdirectory(ui-qt)
endif()
