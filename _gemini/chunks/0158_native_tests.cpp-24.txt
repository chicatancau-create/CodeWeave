  codeweave::execution::ExecutionControl control;
  assert(!control.request(codeweave::execution::ControlCommand::Pause).ok);
  std::atomic_bool cancellation{false};
  codeweave::execution::ProcessResult result;
  const auto working_directory = tool.parent_path();
  std::thread worker([&] {
    result = runner.run({
        {tool, {L"--sleep-ms", L"1200"}, working_directory, "cpp"},
        "source", 3000, 4096}, &cancellation, &control);
  });
  std::this_thread::sleep_for(std::chrono::milliseconds(120));
  assert(control.request(codeweave::execution::ControlCommand::Pause).ok);
  assert(control.is_paused());
  assert(control.request(codeweave::execution::ControlCommand::Step).ok);
  assert(control.is_paused());
  assert(control.request(codeweave::execution::ControlCommand::Resume).ok);
  assert(!control.is_paused());
  cancellation.store(true);
  worker.join();
  assert(result.started && result.cancelled);
  assert(result.control_events.size() == 3);
#else
  (void)tool;
#endif
}

void test_execution_service_compiles_and_runs_cpp(const std::filesystem::path& compiler) {
#ifdef _WIN32
  if (compiler.empty()) return;
  const auto root = std::filesystem::temp_directory_path() /
      ("codeweave-native-cpp-execution-" + std::to_string(std::random_device{}()));
  std::error_code error;
  std::filesystem::create_directories(root, error);
  assert(!error);
  codeweave::execution::ExecutionService service;
  const auto result = service.run(
      {"cpp-e2e-01", "cpp",
       "#include <iostream>\nint main() { std::cout << 42 << '\\n'; }\n",
       10'000, 4096}, compiler, root);
  if (!result.validation.ok || !result.process.has_value()) {
    std::cerr << "cpp compiler execution failed: "
              << (result.validation.message.empty()
                  ? (result.process ? result.process->error_code : "no_process")
                  : result.validation.message)
              << "\n";
  }
  assert(result.validation.ok);
  assert(result.process.has_value());
  assert(result.process->started);
  assert(result.process->exit_code == 0);
  assert(result.process->error_code.empty());
  assert(result.process->output == "42\r\n");
  assert(std::filesystem::is_empty(root, error));
  assert(!error);
  std::filesystem::remove_all(root, error);
#else
  (void)compiler;
#endif
}

void test_cpp_source_trace_preserves_failures_and_bounds(const std::filesystem::path& compiler) {
#ifdef _WIN32
  if (compiler.empty()) return;
  const auto debugger = compiler.parent_path() / "gdb.exe";
  std::error_code error;
  if (!std::filesystem::is_regular_file(debugger, error) || error) return;
  const auto ascii_path = [](const std::filesystem::path& path) {
    const auto text = path.wstring();
    return std::all_of(text.begin(), text.end(), [](wchar_t value) { return value > 0 && value < 128; });
  };
  if (!ascii_path(compiler) || !ascii_path(std::filesystem::temp_directory_path())) return;
  Fixture fixture;
  const auto execution_root = fixture.root / L"中文 execution with spaces";
  assert(std::filesystem::create_directory(execution_root));
  codeweave::execution::ExecutionService service;
  const auto run = [&](std::string source, std::uint32_t timeout_ms = 20'000) {
    const auto result = service.run({"cpp-trace-boundary", "cpp", std::move(source), timeout_ms, 4096},
        compiler, execution_root);
    assert(result.validation.ok && result.process && result.process->started && result.trace);
    assert(std::filesystem::is_empty(execution_root));
    std::size_t count = 0;
    std::size_t bytes = 0;
    for (const auto& event : result.trace->events()) {
      if (event.kind != codeweave::execution::EventKind::SourceStep) continue;
      assert(event.payload.size() <= 8192);
      ++count;
      bytes += event.payload.size();
    }
    assert(count <= 512 && bytes <= 256 * 1024);
    return result;
  };
  const auto success = run("#include <iostream>\nint main() {\n int total = 7;\n total += 2;\n"
