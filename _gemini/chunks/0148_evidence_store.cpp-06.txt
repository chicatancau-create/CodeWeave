      indexed = operation_index::find(index_path(), record.operation_id, index_snapshot_, entry, found);
      if (!indexed.ok || !found) return app::Result::failure("證据 operation 索引与账本不一致");
      const auto reread = operation_index::read_entry(path(), entry, line);
      if (!reread.ok || !parse(line, existing) || existing.operation_id != record.operation_id) {
        return app::Result::failure("证据 operation 索引重建后仍不一致");
      }
    }
    if (!same_record(existing, record)) return app::Result::failure("operationId 已用于不同证据");
    already_present = true;
    return app::Result::success("重复 operationId 已幂等忽略");
  }
  if (index_snapshot_.record_count >= kMaxEvidenceRecords) {
    return app::Result::failure("证据记录数量超过上限");
  }
  if (has_symlink_component(path().parent_path())) {
    return app::Result::failure("证据目录路径不能经过符号链接");
  }
  std::error_code error;
  std::filesystem::create_directories(path().parent_path(), error);
  if (error) return app::Result::failure("创建证据目录失败");
  const auto destination_status = std::filesystem::symlink_status(path(), error);
  if (!error && (std::filesystem::is_symlink(destination_status) ||
      !std::filesystem::is_regular_file(destination_status))) {
    return app::Result::failure("证据账本文件类型无效");
  }
  error.clear();
  std::uintmax_t base_size = 0;
  if (std::filesystem::exists(path(), error)) {
    base_size = std::filesystem::file_size(path(), error);
    if (error) return app::Result::failure("读取证据账本大小失败");
  }
  const auto line = serialize(record) + '\n';
  if (line.size() > kMaxEvidenceLineBytes || base_size > kMaxEvidenceFileBytes - line.size()) {
    return app::Result::failure("证据账本超过大小上限");
  }
  const auto transaction = write_pending_transaction(path(), base_size, line);
  if (!transaction.ok) return transaction;
  const auto appended = append_durable(path(), line);
  if (!appended.ok) return appended;
  index_snapshot_.ledger_size = base_size + line.size();
  const auto indexed_append = operation_index::append(index_path(), path(), data_root_,
      {record.operation_id, base_size, line.size()}, index_snapshot_);
  if (!indexed_append.ok) index_ready_ = false;
  else index_ready_ = true;
  error.clear();
  if (!std::filesystem::remove(pending_path(path()), error) || error) {
    return app::Result::failure("清理证据追加事务失败");
  }
  return app::Result::success("证据已追加保存");
}
