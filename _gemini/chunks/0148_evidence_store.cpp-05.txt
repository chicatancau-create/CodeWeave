  std::lock_guard lock(mutex_);
  if (!is_operation_id(operation_id)) return app::Result::failure("补偿 operationId 无效");
  std::vector<EvidenceRecord> records;
  const auto loaded = read_all_unlocked(path(), data_root_, records);
  if (!loaded.ok) return loaded;
  const auto old_size = records.size();
  records.erase(std::remove_if(records.begin(), records.end(), [operation_id](const auto& item) {
    return item.operation_id == operation_id;
  }), records.end());
  if (records.size() == old_size) return app::Result::success("没有需要补偿的证据");
  index_ready_ = false;
  const auto index_cleared = operation_index::clear(index_path(), data_root_);
  if (!index_cleared.ok) return index_cleared;
  if (records.empty()) {
    std::error_code error;
    if (!std::filesystem::remove(path(), error) && error) return app::Result::failure("删除补偿证据失败");
    return app::Result::success("证据补偿已完成");
  }
  if (has_symlink_component(path().parent_path())) return app::Result::failure("证据目录路径不能经过符号链接");
  auto temporary = path();
  temporary += ".rollback." + std::to_string(std::chrono::steady_clock::now().time_since_epoch().count());
  { std::ofstream output(temporary, std::ios::binary | std::ios::trunc);
    if (!output) return app::Result::failure("创建证据补偿文件失败");
    for (const auto& item : records) output << serialize(item) << '\n';
    output.flush();
    if (!output) { std::error_code ignored; std::filesystem::remove(temporary, ignored); return app::Result::failure("写入证据补偿文件失败"); }
  }
  std::error_code error;
#ifdef _WIN32
  if (!MoveFileExW(temporary.wstring().c_str(), path().wstring().c_str(), MOVEFILE_REPLACE_EXISTING | MOVEFILE_WRITE_THROUGH))
    error = std::error_code(static_cast<int>(GetLastError()), std::system_category());
#else
  std::filesystem::rename(temporary, path(), error);
#endif
  if (error) { std::error_code ignored; std::filesystem::remove(temporary, ignored); return app::Result::failure("提交证据补偿失败"); }
  return app::Result::success("证据补偿已完成");
}

app::Result EvidenceStore::append_idempotent(const EvidenceRecord& record, bool& already_present) {
  std::lock_guard lock(mutex_);
  already_present = false; if (!validate_evidence(record).ok) return app::Result::failure("证据记录无效");
  if (data_root_.empty() || !data_root_.is_absolute()) {
    return app::Result::failure("证据数据根必须是绝对路径");
  }
  const auto recovered = recover_pending(path(), data_root_);
  if (!recovered.ok) return recovered;
  auto parsed = [](std::string_view line, std::string& operation_id) {
    EvidenceRecord record;
    if (!parse(line, record)) return false;
    operation_id = std::move(record.operation_id);
    return true;
  };
  const auto index_status = operation_index::load_or_rebuild(index_path(), path(), data_root_,
      kMaxEvidenceFileBytes, kMaxEvidenceLineBytes, kMaxEvidenceRecords, parsed,
      index_snapshot_);
  if (!index_status.ok) return index_status;
  index_ready_ = true;
  operation_index::Entry entry;
  bool found = false;
  auto indexed = operation_index::find(index_path(), record.operation_id, index_snapshot_, entry, found);
  if (!indexed.ok) {
    const auto rebuilt = operation_index::rebuild(index_path(), path(), data_root_,
        kMaxEvidenceFileBytes, kMaxEvidenceLineBytes, kMaxEvidenceRecords, parsed,
        index_snapshot_);
    if (!rebuilt.ok) return rebuilt;
    indexed = operation_index::find(index_path(), record.operation_id, index_snapshot_, entry, found);
    if (!indexed.ok) return indexed;
  }
  if (found) {
    std::string line;
    const auto read = operation_index::read_entry(path(), entry, line);
    EvidenceRecord existing;
    if (!read.ok || !parse(line, existing) || existing.operation_id != record.operation_id) {
      index_ready_ = false;
      const auto rebuilt = operation_index::rebuild(index_path(), path(), data_root_,
          kMaxEvidenceFileBytes, kMaxEvidenceLineBytes, kMaxEvidenceRecords, parsed,
          index_snapshot_);
      if (!rebuilt.ok) return rebuilt;
