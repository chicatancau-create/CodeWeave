    flex-direction: column;
    /* 地板：窗口太矮时（笔记本横屏 + 浏览器工具栏）不硬挤，退回普通滚动。 */
    min-height: 480px;
    max-height: calc(100vh - var(--cw-chrome-h));
    max-height: calc(100svh - var(--cw-chrome-h));
    /* 一屏化的闸门：工作台自己不许把页面撑长。高度到此为止，多出来的内容由内部
       那几个 `overflow: auto` 的窗格（编辑器 / 控制台 / 变量 / 轨道）各自消化。
       没有这一行，`max-height` 只是个建议——子项撑开就把整页顶下去，学习页照旧要滑。 */
    overflow: hidden;
  }
  /* 页头与指南/洞察是定高附件，中间那两列吃掉剩余高度。 */
  .cw-workspace .learning-header,
  .cw-workspace .lesson-guide,
  .cw-workspace .lesson-insight {
    flex: 0 0 auto;
  }
  .cw-workspace .learning-view {
    flex: 1 1 auto;
    /* 没有它，grid/flex 子项的最小内容尺寸会让两列撑破外层 max-height。 */
    min-height: 0;
    align-items: stretch;
  }
  /* 每列一个滚动条：列内子块不再各自套内层滚动容器（否则一屏里两三个滚动条）。 */
  .cw-workspace .workspace,
  .cw-workspace .right-panels {

    overflow-y: auto;
    overscroll-behavior: contain;
  }
  .cw-workspace .workspace > *,
  .cw-workspace .right-panels > * {
    flex: 0 0 auto;
  }
  /* 放掉列内这些块的自身高度上限：滚动的职责上移到列容器，避免嵌套滚动。 */
  .cw-workspace .reading-pane,
  .cw-workspace .rs-stage,
  .cw-workspace .code-editor,
  .cw-workspace .variable-list {
    max-height: none;
  }
  .cw-workspace .reading-pane {
    overflow: visible;
  }
  /* 指南比一屏还长时给个上限自己滚，别把整页顶下去。这里要再从视口里扣掉
     页头与上面那两列占的地（约 320），否则指南会跟列抢高度、把外层 max-height
     撑破——外层一旦溢出，页面又会长出文档滚动条，等于没做单屏化。 */
  .cw-workspace .lesson-guide {
    max-height: calc(100vh - var(--cw-chrome-h) - 320px);
    max-height: calc(100svh - var(--cw-chrome-h) - 320px);
    overflow-y: auto;
    overscroll-behavior: contain;
  }

/* ── 代码模式：左边写代码，右边看画布（本页「换一种摆法」） ──────────────
   入口是页头那枚 `.cw-mode-toggle`，开合状态挂在根节点的 `.is-code-mode` 上
   （组件见 `learning-workspace-view.tsx` 的 `codeMode`）。**只切呈现，不切 DOM**：
   四个分区锚点与导航 href 表是契约测试逐条钉死的，另起一棵树就要维护两份可达性，
   所以这里一个元素不增不减，靠 CSS 决定此刻谁在场上、摆在哪。

   留场的是「写代码 → 看运行」这条链上的东西：左栏的代码编辑器与运行条，右栏的
   画布、变量、控制台。其余（主题原文、证据/完成、来源明细、课程指南、洞察）整体
   `display: none` 离场——连子树一起离开 tab 顺序与读屏顺序，不留「看不见但 Tab 能到」
   的暗门（`visibility: hidden` 与 `opacity: 0` 都会留下这种暗门）。 */
.cw-workspace.is-code-mode .learning-view {
  /* 与宽屏档同一个两列骨架，只把配比换成画布占多半：参考实现里
     控制栏是窄侧栏、画布是主列，这里反过来让编辑器做窄侧、画布做主场。 */
  grid-template-columns: minmax(0, 1fr) minmax(0, 1.15fr);
}
