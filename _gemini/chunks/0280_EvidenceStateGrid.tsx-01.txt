 * 交上去的完成事实（`payload`）由主编排文件用 `completionPayloadOf()` 现算，这里只负责点这一下。

  isExternal: boolean

  onPatchEvidence: (updates: Partial<ExternalEvidence>) => void

  detailPending: boolean
  canAutoRun: boolean
  /** 六态与「标记已阅读」：全部是上层给的事实与动作。 */
  evidenceStates: LessonEvidenceStates | null
  readMarked: boolean
  onMarkRead?: () => void
}
